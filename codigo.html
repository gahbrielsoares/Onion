1. mp-checkout (atualizado em 20/11, 17h)

// Onion Park · cria a assinatura no Mercado Pago e devolve o link do checkout.
// Também cancela (body: {action:"cancel"}) e confere o pagamento direto no Mercado Pago (body: {action:"sync"}).
// Segredos: MP_ACCESS_TOKEN, APP_URL (ex.: https://gahbrielsoares.github.io/Onion/app/), PRICE (opcional, padrão 29.90)
// COUPONS (opcional, JSON com o preço mensal de cada cupom), ex.: {"BLACK19":19.90,"NOVEM9":9.90}
// Os cupons ficam só no Supabase (o código do site é público).
import { createClient } from "npm:@supabase/supabase-js@2.45.4";

const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" };
// nenhuma chamada ao Mercado Pago pode travar o botão do cliente
const mpFetch = (u: string, init: RequestInit = {}) => fetch(u, { ...init, signal: AbortSignal.timeout(12000) });
const json = (b: unknown, s = 200) => new Response(JSON.stringify(b), { status: s, headers: { ...CORS, "content-type": "application/json" } });

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  try {
    const url = Deno.env.get("SUPABASE_URL")!, service = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
    const MP = Deno.env.get("MP_ACCESS_TOKEN"); if (!MP) return json({ error: "MP_ACCESS_TOKEN não configurado" }, 500);
    const admin = createClient(url, service);
    const jwt = (req.headers.get("authorization") || "").replace("Bearer ", "");
    const { data: { user } } = await admin.auth.getUser(jwt);
    if (!user) return json({ error: "Faça login de novo." }, 401);
    const body = await req.json().catch(() => ({}));
    const { data: sub } = await admin.from("subscriptions").select("*").eq("user_id", user.id).maybeSingle();

    if (body.action === "cancel") {
      if (!sub?.preapproval_id) return json({ error: "Você não tem assinatura ativa." }, 400);
      const r = await mpFetch(`https://api.mercadopago.com/preapproval/${sub.preapproval_id}`, { method: "PUT", headers: { Authorization: `Bearer ${MP}`, "content-type": "application/json" }, body: JSON.stringify({ status: "cancelled" }) });
      if (!r.ok) return json({ error: "O Mercado Pago não aceitou o cancelamento: " + (await r.text()).slice(0, 200) }, 502);
      const until = sub.next_payment_date && new Date(sub.next_payment_date) > new Date() ? sub.next_payment_date : sub.access_until;
      await admin.from("subscriptions").update({ status: "cancelled", access_until: until, updated_at: new Date().toISOString() }).eq("user_id", user.id);
      return json({ ok: true, access_until: until });
    }

    // confere direto no Mercado Pago (não depende do aviso automático chegar)
    if (body.action === "sync") {
      const H = { headers: { Authorization: `Bearer ${MP}` } };
      let list: any[] = [];
      const r = await mpFetch(`https://api.mercadopago.com/preapproval/search?external_reference=${user.id}&limit=50`, H);
      if (r.ok) list = ((await r.json()).results || []).filter((x: any) => x.external_reference === user.id);
      if (sub?.preapproval_id && !list.some((x: any) => x.id === sub.preapproval_id)) {
        const r2 = await mpFetch(`https://api.mercadopago.com/preapproval/${sub.preapproval_id}`, H); if (r2.ok) list.push(await r2.json());
      }
      const pre = list.find((x: any) => x.status === "authorized") || list.find((x: any) => x.id === sub?.preapproval_id);
      if (!pre) return json({ status: sub?.status || "none" });
      const next = pre.next_payment_date || null;
      const row: Record<string, unknown> = { user_id: user.id, preapproval_id: pre.id, status: pre.status, payer_email: pre.payer_email || sub?.payer_email || null, next_payment_date: next, updated_at: new Date().toISOString() };
      if (pre.status === "authorized" && next) row.access_until = next;
      await admin.from("subscriptions").upsert(row);
      return json({ status: pre.status });
    }

    // cupom: o preço vem só do servidor
    let coupons: Record<string, number> = {};
    try { coupons = JSON.parse(Deno.env.get("COUPONS") || "{}"); } catch (_) { coupons = {}; }
    const code = String(body.coupon || "").trim().toUpperCase();
    const base = Number(Deno.env.get("PRICE") || "29.90");
    if (code && !(code in coupons)) return json({ error: "Cupom inválido." }, 400);
    const price = code ? Number(coupons[code]) : base;
    if (body.action === "quote") return json({ price, coupon: code || null });

    if (sub?.status === "authorized") return json({ error: "Sua assinatura já está ativa." }, 400);
    const payer = String(body.payer_email || user.email || "").trim();
    // valores muito baixos podem ser recusados: tenta o valor do cupom e, se não der, o mínimo aceito
    const tries = [...new Set([price, ...(price < 1 ? [1, 2, 5] : [])])];
    let p: any = null, ok = false, used = price;
    for (const amount of tries) {
      const r = await mpFetch("https://api.mercadopago.com/preapproval", {
        method: "POST", headers: { Authorization: `Bearer ${MP}`, "content-type": "application/json" },
        body: JSON.stringify({
          reason: code ? `Onion Park · assinatura mensal (cupom ${code})` : "Onion Park · assinatura mensal",
          external_reference: user.id,
          payer_email: payer,
          back_url: Deno.env.get("APP_URL") || "https://gahbrielsoares.github.io/Onion/app/",
          status: "pending",
          auto_recurring: { frequency: 1, frequency_type: "months", transaction_amount: amount, currency_id: "BRL" }
        })
      });
      p = await r.json().catch(() => ({ message: `resposta inválida (${r.status})` })); ok = r.ok && !!p.init_point; used = amount;
      if (ok) break;
    }
    if (!ok) return json({ error: "O Mercado Pago recusou: " + (p?.message || JSON.stringify(p).slice(0, 200)) }, 502);
    await admin.from("subscriptions").upsert({ user_id: user.id, preapproval_id: p.id, status: "pending", payer_email: payer, updated_at: new Date().toISOString() });
    return json({ init_point: p.init_point, price: used });
  } catch (e) {
    const m = String(e?.message || e);
    if (/timed? ?out|abort/i.test(m)) return json({ error: "O Mercado Pago demorou para responder. Tente de novo em instantes." }, 504);
    return json({ error: m }, 500);
  }
});

2. mp-webhook (atualizado em 20/11, 17h)

// Onion Park · recebe os avisos do Mercado Pago e atualiza a assinatura.
// Publique com "Verify JWT" DESLIGADO (o Mercado Pago não manda login).
// Segredos: MP_ACCESS_TOKEN, MP_WEBHOOK_SECRET (assinatura secreta das notificações)
import { createClient } from "npm:@supabase/supabase-js@2.45.4";

async function hmac(secret: string, msg: string) {
  const k = await crypto.subtle.importKey("raw", new TextEncoder().encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  return [...new Uint8Array(await crypto.subtle.sign("HMAC", k, new TextEncoder().encode(msg)))].map(b => b.toString(16).padStart(2, "0")).join("");
}

Deno.serve(async (req) => {
  const u = new URL(req.url);
  const body = await req.json().catch(() => ({} as Record<string, unknown>));
  const type = String((body as any).type || u.searchParams.get("type") || u.searchParams.get("topic") || "");
  const id = String((body as any)?.data?.id || u.searchParams.get("data.id") || u.searchParams.get("id") || "");
  if (!id) return new Response("ok");

  // confere a assinatura secreta (x-signature) quando o segredo estiver configurado
  const secret = Deno.env.get("MP_WEBHOOK_SECRET");
  if (secret) {
    const sig = req.headers.get("x-signature") || "", reqId = req.headers.get("x-request-id") || "";
    const parts = Object.fromEntries(sig.split(",").map(s => s.trim().split("=")));
    const manifest = `id:${(u.searchParams.get("data.id") || id).toLowerCase()};request-id:${reqId};ts:${parts.ts};`;
    // se não bater, só registra: os dados abaixo vêm sempre da API do Mercado Pago com o seu token,
    // então um aviso falso não consegue liberar ninguém; e um segredo digitado errado não trava os clientes
    if (!parts.v1 || (await hmac(secret, manifest)) !== parts.v1) console.log("x-signature não conferiu; seguindo com a consulta à API");
  }

  const MP = Deno.env.get("MP_ACCESS_TOKEN")!;
  const admin = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
  let pre: any = null;
  if (/preapproval/.test(type)) {
    const r = await fetch(`https://api.mercadopago.com/preapproval/${id}`, { headers: { Authorization: `Bearer ${MP}` } }); if (r.ok) pre = await r.json();
  } else if (/authorized_payment/.test(type)) {
    const r = await fetch(`https://api.mercadopago.com/authorized_payments/${id}`, { headers: { Authorization: `Bearer ${MP}` } });
    if (r.ok) { const ap = await r.json(); if (ap.preapproval_id) { const r2 = await fetch(`https://api.mercadopago.com/preapproval/${ap.preapproval_id}`, { headers: { Authorization: `Bearer ${MP}` } }); if (r2.ok) pre = await r2.json(); } }
  }
  if (!pre || !pre.external_reference) return new Response("ok");

  // um aviso de uma assinatura antiga (abandonada ou trocada) não pode derrubar a assinatura atual
  const { data: cur } = await admin.from("subscriptions").select("preapproval_id,status").eq("user_id", pre.external_reference).maybeSingle();
  if (cur && cur.preapproval_id && cur.preapproval_id !== pre.id && pre.status !== "authorized") return new Response("ok");

  const next = pre.next_payment_date || null;
  const row: Record<string, unknown> = { user_id: pre.external_reference, preapproval_id: pre.id, status: pre.status, payer_email: pre.payer_email || null, next_payment_date: next, updated_at: new Date().toISOString() };
  if (pre.status === "authorized" && next) row.access_until = next;   // pago até a próxima cobrança
  await admin.from("subscriptions").upsert(row);
  return new Response("ok");
});
